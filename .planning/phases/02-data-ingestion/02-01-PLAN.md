    object ConfidenceRouter {
        val tier1 = setOf("com.whoop.android", "com.ouraring.oura") // Tier 1: Dedicated High-Precision
        val tier2 = setOf("com.garmin.android.apps.connectmobile", "com.fitbit.FitbitMobile", "com.coros.wearable") // Tier 2: Fitness Watches
        val tier3 = setOf("com.samsung.android.app.shealth", "com.google.android.apps.fitness") // Tier 3: Smartwatches
        val tier4 = setOf("com.aira.health.mock") // Tier 4: Phone-Only Fallback / Unknown

        fun getConfidenceWeight(packageName: String): Int {
            return when (packageName) {
                in tier1 -> 100
                in tier2 -> 85
                in tier3 -> 65
                else -> 40
            }
        }
    }

    object HealthDataModule {
        @Provides
        @Singleton
        fun provideHealthConnectClient(@ApplicationContext context: Context): HealthConnectClient? {
            return if (HealthConnectClient.getSdkStatus(context) == HealthConnectClient.SDK_AVAILABLE) {
                HealthConnectClient.getOrCreate(context)

                null
            }
